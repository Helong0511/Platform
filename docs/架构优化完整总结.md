```

├── services/
│   ├── base_service.py          # 服务基类
│   ├── user_service.py          # 用户服务

│       ├── offer_service.py
│       └── conversion_service.py

```
app/
├── repositories/
│   ├── base_repository.py       # Repository基类
│   └── offer_repository.py      # Offer Repository示例
