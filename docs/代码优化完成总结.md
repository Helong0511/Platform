### 2.2 代码对比

#### 之前（重复代码）：
```javascript
// 错误处理重复
catch (error) {
  if (isErrorHandled(error)) {
    throw error;
  }
  const errorMsg = extractErrorMessage(error);
  ElMessage.error(errorMsg);
  throw error;
}

// CSRF Token刷新重复
const { refreshCsrfToken } = await import('../utils/api');
await refreshCsrfToken('/api/v1/users/stats');
await ensureCsrfToken();

// 文件下载重复
const blob = new Blob([response.data]);
const url = window.URL.createObjectURL(blob);
const link = document.createElement('a');
link.href = url;
// ... 大量重复代码
```

#### 现在（使用公共函数）：
```javascript
// 错误处理简化
catch (error) {
  const { handleError } = await import('../utils/api');
  handleError(error);
}

// CSRF Token刷新简化
const { refreshCsrfTokenBeforeRequest } = await import('../utils/api');
await refreshCsrfTokenBeforeRequest('/api/v1/users/stats');

// 文件下载简化
const { handleFileDownload } = await import('../utils/api');
handleFileDownload(response, '文件名.xlsx', () => {
  ElMessage.success('导出成功');
});
```
