### 2.1 现有表结构（保持不变）

#### 2.1.1 products 表（产品表）
```sql
CREATE TABLE `products` (
  `id` INT(11) NOT NULL AUTO_INCREMENT COMMENT '产品ID',
  `client_id` INT(11) NOT NULL COMMENT '客户ID',
  `name` VARCHAR(100) NOT NULL COMMENT '产品名称',
  `ecpm_deduct_ratio` FLOAT NOT NULL DEFAULT 0.0 COMMENT 'eCPM减扣比例',
  `revenue_deduct_ratio` FLOAT NOT NULL DEFAULT 0.0 COMMENT '收益减扣比例',
  `is_active` TINYINT(1) NOT NULL DEFAULT 1 COMMENT '是否激活',
  `created_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`),
  KEY `idx_products_client` (`client_id`),
  KEY `idx_products_name` (`name`),
  FULLTEXT KEY `ft_products_name` (`name`) WITH PARSER `ngram`,
  CONSTRAINT `fk_products_client` FOREIGN KEY (`client_id`) REFERENCES `clients`(`id`) ON DELETE RESTRICT ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='产品表';
```

#### 2.1.3 ad_data 表（广告数据表）
```sql
CREATE TABLE `ad_data` (
  `id` INT(11) NOT NULL AUTO_INCREMENT COMMENT '广告数据ID',
  `date` DATETIME NOT NULL COMMENT '数据日期',
  `product_id` INT(11) NOT NULL COMMENT '产品ID',
  `request_count` INT NOT NULL DEFAULT 0 COMMENT '请求数',
  `match_count` INT NOT NULL DEFAULT 0 COMMENT '匹配数',
  `display_count` INT NOT NULL DEFAULT 0 COMMENT '展示数',
  `click_count` INT NOT NULL DEFAULT 0 COMMENT '点击数',
  `ecpm_before_deduct` FLOAT NOT NULL DEFAULT 0.0 COMMENT 'eCPM未减扣',
  `ecpm_after_deduct` FLOAT NOT NULL DEFAULT 0.0 COMMENT 'eCPM已减扣',
  `revenue_before_deduct` FLOAT NOT NULL DEFAULT 0.0 COMMENT '收益未减扣',
  `revenue_after_deduct` FLOAT NOT NULL DEFAULT 0.0 COMMENT '收益已减扣',
  
  -- 计算字段（生成列）
  `coverage_rate` FLOAT GENERATED ALWAYS AS (
    CASE 
      WHEN `request_count` > 0 THEN ROUND(`match_count` / `request_count` * 100, 2)
      ELSE 0
    END
  ) STORED COMMENT '覆盖率 = 匹配数 / 请求数 * 100',
  `display_rate` FLOAT GENERATED ALWAYS AS (
    CASE 
      WHEN `match_count` > 0 THEN ROUND(`display_count` / `match_count` * 100, 2)
      ELSE 0
    END
  ) STORED COMMENT '展示率 = 展示数 / 匹配数 * 100',
  `click_rate` FLOAT GENERATED ALWAYS AS (
    CASE 
      WHEN `display_count` > 0 THEN ROUND(`click_count` / `display_count` * 100, 2)
      ELSE 0
    END
  ) STORED COMMENT '点击率 = 点击数 / 展示数 * 100',
  
  `created_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_ad_data_date_product` (`date`, `product_id`),
  KEY `idx_ad_data_date` (`date`),
  KEY `idx_ad_data_product` (`product_id`),
  KEY `idx_ad_data_date_product` (`date`, `product_id`),
  CONSTRAINT `fk_ad_product` FOREIGN KEY (`product_id`) REFERENCES `products`(`id`) ON DELETE RESTRICT ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='广告数据表';
```

### 2.2 新增表结构

#### 2.2.1 product_change_history 表（产品变更历史表）
**目的**：详细记录产品字段变更历史，包括字段名、旧值、新值、变更原因等

```sql
CREATE TABLE `product_change_history` (
  `id` INT(11) NOT NULL AUTO_INCREMENT COMMENT '变更记录ID',
  `product_id` INT(11) NOT NULL COMMENT '产品ID',
  `user_id` INT(11) DEFAULT NULL COMMENT '操作用户ID',
  `username` VARCHAR(50) DEFAULT NULL COMMENT '操作用户名（冗余字段）',
  `change_type` VARCHAR(50) NOT NULL COMMENT '变更类型：CREATE, UPDATE, DELETE, TRANSFER, ACTIVATE, DEACTIVATE',
  `field_name` VARCHAR(50) DEFAULT NULL COMMENT '变更字段名（如：name, ecpm_deduct_ratio, client_id等）',
  `old_value` TEXT DEFAULT NULL COMMENT '旧值（JSON格式，支持复杂对象）',
  `new_value` TEXT DEFAULT NULL COMMENT '新值（JSON格式，支持复杂对象）',
  `change_reason` VARCHAR(500) DEFAULT NULL COMMENT '变更原因',
  `ip_address` VARCHAR(50) DEFAULT NULL COMMENT '操作IP地址',
  `user_agent` VARCHAR(500) DEFAULT NULL COMMENT '用户代理',
  `request_path` VARCHAR(500) DEFAULT NULL COMMENT '请求路径',
  `created_at` DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) COMMENT '变更时间',
  PRIMARY KEY (`id`),
  KEY `idx_pch_product` (`product_id`),
  KEY `idx_pch_user` (`user_id`),
  KEY `idx_pch_change_type` (`change_type`),
  KEY `idx_pch_field_name` (`field_name`),
  KEY `idx_pch_created_at` (`created_at`),
  KEY `idx_pch_product_change_type` (`product_id`, `change_type`),
  KEY `idx_pch_product_created_at` (`product_id`, `created_at`),
  CONSTRAINT `fk_pch_product` FOREIGN KEY (`product_id`) REFERENCES `products`(`id`) ON DELETE CASCADE ON UPDATE RESTRICT,
  CONSTRAINT `fk_pch_user` FOREIGN KEY (`user_id`) REFERENCES `users`(`id`) ON DELETE SET NULL ON UPDATE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='产品变更历史表';
```

**索引说明**：
- `idx_pch_product`: 按产品ID查询变更历史
- `idx_pch_user`: 按用户查询变更记录
- `idx_pch_change_type`: 按变更类型过滤
- `idx_pch_field_name`: 按字段名过滤
- `idx_pch_created_at`: 按时间排序
- `idx_pch_product_change_type`: 复合索引，优化"查询某产品的特定类型变更"
- `idx_pch_product_created_at`: 复合索引，优化"查询某产品的时间序列变更"

## 六、技术实现细节

### 6.1 变更历史记录实现

#### 6.1.1 在更新产品时记录变更
```python
async def record_product_change(
    db: AsyncSession,
    product_id: int,
    user: User,
    change_type: str,
    field_changes: dict,  # {field_name: {"old": old_value, "new": new_value}}
    change_reason: str = None,
    request: Request = None
):
    """记录产品变更历史"""
    for field_name, values in field_changes.items():
        change_history = ProductChangeHistory(
            product_id=product_id,
            user_id=user.id,
            username=user.username,
            change_type=change_type,
            field_name=field_name,
            old_value=json.dumps(values.get("old"), ensure_ascii=False),
            new_value=json.dumps(values.get("new"), ensure_ascii=False),
            change_reason=change_reason,
            ip_address=request.client.host if request else None,
            user_agent=request.headers.get("user-agent") if request else None,
            request_path=request.url.path if request else None
        )
        db.add(change_history)
```

#### 6.1.2 在创建产品时记录变更
```python
# 在 create_product 中调用

        "ecpm_deduct_ratio": {"old": None, "new": product.ecpm_deduct_ratio},

)
```

### 6.2 产品统计信息实现

#### 6.2.1 实时查询方式（推荐，如果数据量不大）
```python
async def get_product_statistics(
    db: AsyncSession,
    product_id: int
) -> dict:
    """获取产品统计信息（实时查询）"""
    # 查询广告数据汇总
    stats_query = select(
        func.count(AdData.id).label("total_count"),
        func.min(AdData.date).label("first_date"),
        func.max(AdData.date).label("last_date"),
        func.sum(AdData.request_count).label("total_request"),
        func.sum(AdData.match_count).label("total_match"),
        func.sum(AdData.display_count).label("total_display"),
        func.sum(AdData.click_count).label("total_click"),
        func.sum(AdData.revenue_after_deduct).label("total_revenue"),
        func.avg(AdData.ecpm_after_deduct).label("avg_ecpm")
    ).where(AdData.product_id == product_id)
    
    result = await db.execute(stats_query)
    stats = result.first()
    
    return {
        "total_ad_data_count": stats.total_count or 0,
        "first_data_date": stats.first_date,
        "last_data_date": stats.last_date,
        "total_request_count": stats.total_request or 0,
        "total_match_count": stats.total_match or 0,
        "total_display_count": stats.total_display or 0,
        "total_click_count": stats.total_click or 0,
        "total_revenue_after_deduct": float(stats.total_revenue or 0),
        "avg_ecpm_after_deduct": float(stats.avg_ecpm or 0)
    }
```

### A.3 初始化现有产品的变更历史
```sql
-- 为现有产品创建初始变更记录
INSERT INTO `product_change_history` (
    `product_id`, 
    `change_type`, 
    `field_name`, 
    `old_value`, 
    `new_value`,
    `created_at`
)
SELECT 
    `id` AS `product_id`,
    'CREATE' AS `change_type`,
    'initial' AS `field_name`,
    NULL AS `old_value`,
    JSON_OBJECT(
        'name', `name`,
        'client_id', `client_id`,
        'ecpm_deduct_ratio', `ecpm_deduct_ratio`,
        'revenue_deduct_ratio', `revenue_deduct_ratio`,
        'is_active', `is_active`
    ) AS `new_value`,
    `created_at`
FROM `products`
WHERE NOT EXISTS (
    SELECT 1 FROM `product_change_history` 
    WHERE `product_change_history`.`product_id` = `products`.`id`
    AND `product_change_history`.`change_type` = 'CREATE'
);
```
