```javascript
const fetchFilterProducts = async (clientId) => {
  // 获取当前日期范围
  let startDate = null;
  let endDate = null;
  if (searchForm.value.dateRange && searchForm.value.dateRange.length === 2) {
    startDate = searchForm.value.dateRange[0];
    endDate = searchForm.value.dateRange[1];
  }
  // 如果没有选择日期范围，后端会默认使用本月数据
  
  // 调用API获取产品列表
  // clientId = null → 全部客户的产品
  // clientId = 具体值 → 该客户的产品
  const result = await getProductsWithData(clientId || null, startDate, endDate);
  
  // 按名称排序并添加"全部产品"选项
  filterProducts.value = [
    { label: '全部产品', value: '' },
    ...productOptions.sort((a, b) => a.label.localeCompare(b.label, 'zh-CN'))
  ];
};
```
✅ **逻辑**: 正确实现

#### 3. 客户变化处理
```javascript
const handleFilterClientChange = async (clientId) => {
  searchForm.value.productId = '';  // 清空产品选择
  await fetchFilterProducts(clientId || null);  // 重新加载产品列表
};
```
✅ **逻辑**: 客户变化时重新加载产品列表

#### 4. 日期范围变化处理

```javascript
const handleDateRangeChange = async () => {
```

```python
):
    # 如果没有指定日期范围，默认查询当前月的数据
    if not start_date and not end_date:
        today = date.today()
        month_start = date(today.year, today.month, 1)
        _, last_day = monthrange(today.year, today.month)
        month_end = date(today.year, today.month, last_day)
```
