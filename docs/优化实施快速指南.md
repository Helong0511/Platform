### 1.1 添加查询缓存工具类

**创建文件：** `app/core/cache.py`

```python
"""
缓存工具模块
提供查询结果缓存功能
"""
from app.core.redis_client import get_redis_client
import json
import hashlib
import logging
from typing import Any, Optional

logger = logging.getLogger(__name__)

CACHE_PREFIX = "query_cache:"
DEFAULT_TTL = 300  # 默认 5 分钟


def generate_cache_key(*args, **kwargs) -> str:
    """生成缓存键"""
    key_str = json.dumps({"args": args, "kwargs": kwargs}, sort_keys=True, default=str)
    return hashlib.md5(key_str.encode()).hexdigest()


async def get_cached(key: str) -> Optional[Any]:
    """获取缓存"""
    redis = await get_redis_client()
    if not redis:
        return None
    
    try:
        cached = await redis.get(f"{CACHE_PREFIX}{key}")
        if cached:
            return json.loads(cached)
    except Exception as e:
        logger.warning(f"获取缓存失败: {e}")
    return None


async def set_cached(key: str, value: Any, ttl: int = DEFAULT_TTL) -> bool:
    """设置缓存"""
    redis = await get_redis_client()
    if not redis:
        return False
    
    try:
        await redis.setex(
            f"{CACHE_PREFIX}{key}",
            ttl,
            json.dumps(value, default=str)
        )
        return True
    except Exception as e:
        logger.warning(f"设置缓存失败: {e}")
        return False


async def invalidate_cache(pattern: str) -> int:
    """清除匹配模式的缓存"""
    redis = await get_redis_client()
    if not redis:
        return 0
    
    try:
        keys = await redis.keys(f"{CACHE_PREFIX}{pattern}*")
        if keys:
            return await redis.delete(*keys)
    except Exception as e:
        logger.warning(f"清除缓存失败: {e}")
    return 0


def cache_query(ttl: int = DEFAULT_TTL):
    """查询缓存装饰器"""
    def decorator(func):
        async def wrapper(*args, **kwargs):
            # 生成缓存键
            cache_key = generate_cache_key(func.__name__, *args, **kwargs)
            
            # 尝试从缓存获取
            cached = await get_cached(cache_key)
            if cached is not None:
                return cached
            
            # 执行查询
            result = await func(*args, **kwargs)
            
            # 缓存结果
            await set_cached(cache_key, result, ttl)
            
            return result
        return wrapper
    return decorator
```

### 1.3 添加日期查询工具函数

**创建文件：** `app/core/query_utils.py`

```python
"""
查询工具模块
提供常用的查询优化函数
"""
from datetime import date, datetime
from sqlalchemy import Column, and_
from typing import Optional


def date_range_condition(
    column: Column,
    start_date: Optional[date] = None,
    end_date: Optional[date] = None
) -> Optional[any]:
    """
    生成日期范围查询条件（优化版本）
    避免在列上使用函数，提升索引使用率
    """
    conditions = []
    
    if start_date:
        start_datetime = datetime.combine(start_date, datetime.min.time())
        conditions.append(column >= start_datetime)
    
    if end_date:
        end_datetime = datetime.combine(end_date, datetime.max.time())
        conditions.append(column <= end_datetime)
    
    return and_(*conditions) if conditions else None


def product_assignment_join_condition(
    ad_data_date: Column,
    assignment_start: Column,
    assignment_end: Column
) -> any:
    """
    生成产品分配历史 JOIN 条件（优化版本）
    避免在 ad_data.date 上使用 func.date()
    """
    from sqlalchemy import or_, text
    
    return and_(
        ad_data_date >= assignment_start,
        or_(
            assignment_end.is_(None),
            ad_data_date < text("DATE_ADD(assignment_end, INTERVAL 1 DAY)")
        )
    )
```

```python
async def batch_upsert_ad_data(
    data_list: list[dict],
    db: AsyncSession,
    batch_size: int = 1000

    """批量插入/更新广告数据"""
    for i in range(0, len(data_list), batch_size):
```

### 2.2 数据归档脚本

**创建文件：** `app/scheduler_module/tasks/archive_old_data.py`

```python
"""
数据归档任务
定期归档旧数据到归档表
"""
from sqlalchemy import text
from datetime import datetime, timedelta
from app.core.database import AsyncSessionLocal
import logging

logger = logging.getLogger(__name__)


async def archive_old_ad_data(months: int = 6):
    """
    归档指定月数之前的数据
    """
    async with AsyncSessionLocal() as db:
        try:
            # 计算归档日期
            archive_date = datetime.now() - timedelta(days=months * 30)
            
            # 检查归档表是否存在
            check_table = text("""
                SELECT COUNT(*) as count 
                FROM information_schema.tables 
                WHERE table_schema = DATABASE() 
                AND table_name = 'ad_data_archive'
            """)
            result = await db.execute(check_table)
            table_exists = result.scalar() > 0
            
            if not table_exists:
                # 创建归档表
                create_table = text("""
                    CREATE TABLE `ad_data_archive` LIKE `ad_data`;
                    ALTER TABLE `ad_data_archive` ENGINE=ARCHIVE;
                """)
                await db.execute(create_table)
                await db.commit()
                logger.info("创建归档表成功")
            
            # 归档数据
            archive_sql = text("""
                INSERT INTO `ad_data_archive` 
                SELECT * FROM `ad_data` 
                WHERE `date` < :archive_date
            """)
            
            delete_sql = text("""
                DELETE FROM `ad_data` 
                WHERE `date` < :archive_date
            """)
            
            # 先插入到归档表
            result = await db.execute(archive_sql, {"archive_date": archive_date})
            archived_count = result.rowcount
            
            # 再删除原表数据
            await db.execute(delete_sql, {"archive_date": archive_date})
            await db.commit()
            
            logger.info(f"归档完成：归档 {archived_count} 条数据")
            return archived_count
            
        except Exception as e:
            await db.rollback()
            logger.error(f"归档失败: {e}", exc_info=True)
            raise
```

```python
检查 N+1 查询问题

import asyncio
from sqlalchemy import select
from app.core.database import AsyncSessionLocal
from app.models.models import Product, Client

async def check_n_plus_one():
    async with AsyncSessionLocal() as db:
        # 检查产品查询
        products = await db.execute(select(Product).limit(10))
        for product in products.scalars():
            # 如果这里会触发额外查询，就是 N+1 问题
            print(f"Product: {product.name}, Client: {product.client.company_name}")
```

```python
from sqlalchemy.orm import selectinload

products = await db.execute(
    select(Product)
    .options(selectinload(Product.client))
    .limit(10)
)
```

### 3.2 慢查询监控

**创建监控脚本：** `scripts/monitor_slow_queries.py`

```python
"""
监控慢查询
"""
import asyncio
from sqlalchemy import event, text
from app.core.database import engine
import logging

logger = logging.getLogger(__name__)

@event.listens_for(engine.sync_engine, "before_cursor_execute")
def receive_before_cursor_execute(conn, cursor, statement, parameters, context, executemany):
    conn.info.setdefault('query_start_time', []).append(time.time())

@event.listens_for(engine.sync_engine, "after_cursor_execute")
def receive_after_cursor_execute(conn, cursor, statement, parameters, context, executemany):
    total = time.time() - conn.info['query_start_time'].pop(-1)
    if total > 1.0:  # 超过 1 秒
        logger.warning(f"Slow query ({total:.2f}s): {statement[:200]}")
```

```python
"""
性能测试脚本
"""
import asyncio
import time
from app.core.database import AsyncSessionLocal
from app.models.models import AdData
from sqlalchemy import select, func
from datetime import date, timedelta

async def test_query_performance():
    async with AsyncSessionLocal() as db:
        # 测试日期范围查询
        start_date = date.today() - timedelta(days=30)
        end_date = date.today()
        
        start_time = time.time()
        result = await db.execute(
            select(func.count(AdData.id))
            .where(AdData.date >= start_date)
            .where(AdData.date <= end_date)

        elapsed = time.time() - start_time
        
        print(f"查询 {count} 条记录，耗时 {elapsed:.2f} 秒")

if __name__ == "__main__":
    asyncio.run(test_query_performance())
```

```python
# 添加性能监控中间件
class PerformanceMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        start_time = time.time()
        response = await call_next(request)
        process_time = time.time() - start_time
        
        # 记录慢请求
        if process_time > 1.0:
            logger.warning(
                f"Slow request: {request.method} {request.url.path} "
                f"took {process_time:.2f}s"
            )
        
        response.headers["X-Process-Time"] = str(process_time)
        return response
```
