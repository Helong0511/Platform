# 实际重构示例：用户列表Endpoint

## 📋 重构目标

将 `app/api/api_v1/endpoints/users.py` 中的 `get_users` endpoint 重构为使用新架构模式。

## 🔄 重构对比

### 重构前（当前代码）

```python
@router.get("/", response_model=PaginatedResponseBaseSchema[list[UserResponse]], summary="获取用户列表")
async def get_users(
        request: Request,
        db: AsyncSession = Depends(get_db),
        current_user: User = Depends(require_admin),
        page: int = Query(1, ge=1, description="页码"),
        page_size: int = Query(10, ge=1, le=MAX_PAGE_SIZE, description="每页条数"),
        username: str | None = Query(None, description="用户名过滤"),
        role: str | None = Query(None, description="角色过滤"),
        status: str | None = Query(None, description="状态过滤: active/inactive")
):
    """
    获取用户列表，支持分页和过滤
    - 仅管理员可访问
    - 支持按用户名、角色和状态过滤
    - 返回分页用户列表
    - 根据设备类型自动优化分页大小
    """
    # ====== 设备类型优化：自动调整分页大小 ======
    from app.core.utils import optimize_page_size_for_device
    page_size = optimize_page_size_for_device(request, page_size, default_page_size=10)
    
    # ====== 构建查询 ======
    query = select(User)

    # 用户名搜索（优先使用全文搜索，回退到LIKE）
    if username:
        from app.core.fulltext_search import search_with_fulltext
        query = await search_with_fulltext(
            db=db,
            model_class=User,
            search_fields=["username", "full_name", "email"],
            search_term=username,
            base_query=query,
            use_fulltext=True
        )

    # 角色精确匹配（处理空字符串）
    role_filter = None
    if role and role.strip():
        try:
            role_filter = UserRole(role)
        except ValueError:
            logger.warning(f"Invalid role filter value: {role}")
            role_filter = None

    if role_filter:
        query = query.where(User.role == role_filter)

    # 状态过滤（处理空字符串）
    status_filter = None
    if status and status.strip():
        if status.lower() == 'active':
            status_filter = True
        elif status.lower() == 'inactive':
            status_filter = False

    if status_filter is not None:
        query = query.where(User.is_active == status_filter)

    # ====== 计算分页参数 ======
    offset = (page - 1) * page_size

    # ====== 执行查询获取总数 ======
    count_query = select(func.count(User.id))
    if username:
        from app.core.fulltext_search import search_with_fulltext
        count_query = await search_with_fulltext(
            db=db,
            model_class=User,
            search_fields=["username", "full_name", "email"],
            search_term=username,
            base_query=count_query,
            use_fulltext=True
        )
    if role_filter:
        count_query = count_query.where(User.role == role_filter)
    if status_filter is not None:
        count_query = count_query.where(User.is_active == status_filter)

    total_result = await db.execute(count_query)
    total = total_result.scalar() or 0

    result = await db.execute(query.order_by(User.id).offset(offset).limit(page_size))
    users = result.scalars().unique().all()

    # 批量获取客户信息（避免N+1查询）
    client_user_ids = [user.id for user in users if user.role == UserRole.CLIENT]
    client_map = {}
    if client_user_ids:
        try:
            client_query = select(Client).where(Client.user_id.in_(client_user_ids))
            client_result = await db.execute(client_query)
            clients = client_result.scalars().all()
            client_map = {client.user_id: client for client in clients}
        except Exception as e:
            logger.error(f"查询客户信息失败: {str(e)}", exc_info=True)
            client_map = {}

    # 转换为响应模型并添加客户信息
    user_responses = []
    for user in users:
        client_data = None
        api_enabled = False
        password_change_enabled = False
        if user.role == UserRole.CLIENT:
            client = client_map.get(user.id)
            if client:
                api_enabled = bool(client.api_enabled) if client.api_enabled is not None else False
                password_change_enabled = bool(client.password_change_enabled) if client.password_change_enabled is not None else False
                client_data = {
                    'id': client.id,
                    'company_name': client.company_name,
                    'contact_person': client.contact_person,
                    'phone': client.phone,
                    'address': client.address,
                    'is_active': client.is_active,
                    'api_enabled': api_enabled,
                    'password_change_enabled': password_change_enabled
                }
        
        user_dict = {
            'id': user.id,
            'username': user.username,
            'email': user.email,
            'full_name': user.full_name,
            'role': user.role.value,
            'is_active': user.is_active,
            'created_at': user.created_at,
            'updated_at': user.updated_at,
            'api_enabled': api_enabled,
            'password_change_enabled': password_change_enabled,
            'client': client_data
        }
        user_responses.append(UserResponse(**user_dict))

    return PaginatedResponseBaseSchema.success(
        data=user_responses,
        total=total,
        page=page,
        page_size=page_size
    )
```

**代码行数**：约 150 行  
**问题**：
- 业务逻辑在endpoint中
- 查询逻辑复杂且重复
- 手动处理客户信息关联
- 代码难以测试和维护

### 重构后（使用新架构）

```python
from app.core.decorators import require_admin, validate_pagination
from app.core.dependencies import UserServiceDep
from app.core.response import create_paginated_response
from app.core.utils import optimize_page_size_for_device

@router.get("/", response_model=PaginatedResponseBaseSchema[list[UserResponse]], summary="获取用户列表")
@require_admin  # 使用装饰器进行权限检查
@validate_pagination(max_page_size=MAX_PAGE_SIZE)  # 使用装饰器验证分页参数
async def get_users(
        request: Request,
        page: int = Query(1, ge=1, description="页码"),
        page_size: int = Query(10, ge=1, le=MAX_PAGE_SIZE, description="每页条数"),
        username: str | None = Query(None, description="用户名过滤"),
        role: str | None = Query(None, description="角色过滤"),
        status: str | None = Query(None, description="状态过滤: active/inactive"),
        user_service: UserServiceDep,  # 使用服务依赖注入
        current_user: User = Depends(get_current_user)
):
    """
    获取用户列表，支持分页和过滤（重构后版本）
    
    重构说明：
    - 使用装饰器进行权限检查和参数验证
    - 业务逻辑在服务层（UserService.get_users）
    - 使用统一响应格式
    - 代码更简洁，易于维护
    """
    # 设备类型优化（保留在endpoint中，因为需要Request对象）
    page_size = optimize_page_size_for_device(request, page_size, default_page_size=10)
    
    # 状态转换
    is_active = None
    if status:
        if status.lower() == 'active':
            is_active = True
        elif status.lower() == 'inactive':
            is_active = False
    
    # 调用服务层（业务逻辑在服务层）
    users, total = await user_service.get_users(
        current_user=current_user,
        page=page,
        page_size=page_size,
        username=username,
        role=role,
        is_active=is_active,
        search=username  # 使用search参数
    )
    
    # 转换为响应模型（处理客户信息）
    user_responses = []
    for user in users:
        client_data = None
        api_enabled = False
        password_change_enabled = False
        
        # 客户信息已通过关联预加载，直接访问
        if user.role == UserRole.CLIENT and user.clients:
            client = user.clients[0]  # 客户用户通常只有一个客户
            api_enabled = bool(client.api_enabled) if client.api_enabled is not None else False
            password_change_enabled = bool(client.password_change_enabled) if client.password_change_enabled is not None else False
            client_data = {
                'id': client.id,
                'company_name': client.company_name,
                'contact_person': client.contact_person,
                'phone': client.phone,
                'address': client.address,
                'is_active': client.is_active,
                'api_enabled': api_enabled,
                'password_change_enabled': password_change_enabled
            }
        
        user_dict = {
            'id': user.id,
            'username': user.username,
            'email': user.email,
            'full_name': user.full_name,
            'role': user.role.value,
            'is_active': user.is_active,
            'created_at': user.created_at,
            'updated_at': user.updated_at,
            'api_enabled': api_enabled,
            'password_change_enabled': password_change_enabled,
            'client': client_data
        }
        user_responses.append(UserResponse(**user_dict))

    # 使用统一响应格式
    return create_paginated_response(
        items=user_responses,
        total=total,
        page=page,
        page_size=page_size
    )
```

**代码行数**：约 70 行（减少 53%）  
**优势**：
- ✅ 业务逻辑在服务层，可复用
- ✅ 使用装饰器简化权限检查
- ✅ 使用服务依赖注入
- ✅ 使用统一响应格式
- ✅ 代码更简洁，易于维护

## 🔧 服务层实现（UserService.get_users）

```python
async def get_users(
    self,
    current_user: User,
    page: int = 1,
    page_size: int = 10,
    username: Optional[str] = None,
    role: Optional[str] = None,
    is_active: Optional[bool] = None,
    search: Optional[str] = None
) -> tuple[List[User], int]:
    """
    获取用户列表（支持分页、筛选、搜索）
    
    使用新的查询构建器和仓储层，代码更简洁，性能更优
    """
    # 权限检查：仅管理员可查看所有用户
    self.require_admin(current_user)
    
    # 构建过滤条件
    filters = {}
    if is_active is not None:
        filters["is_active"] = is_active
    
    if role:
        try:
            role_enum = UserRole(role)
            filters["role"] = role_enum
        except ValueError:
            pass  # 无效的角色值，忽略
    
    # 使用仓储层的分页查询（通过查询构建器）
    # 如果提供了username，使用search参数；否则使用search参数（如果提供）
    search_keyword = username or search
    
    return await self.repository.paginated(
        page=page,
        page_size=page_size,
        filters=filters,
        search_fields=["username", "full_name", "email"],
        search_keyword=search_keyword,
        order_by_field="created_at",
        desc_order=True,
        load_relations=["clients"]  # 预加载客户信息，避免N+1查询
    )
```

## ✨ 关键改进点

1. **权限检查**：从手动检查改为装饰器 `@require_admin`
2. **参数验证**：从手动验证改为装饰器 `@validate_pagination`
3. **业务逻辑**：从endpoint移到服务层
4. **查询构建**：从手动构建改为使用查询构建器
5. **关联加载**：从手动批量查询改为自动预加载
6. **响应格式**：从手动构建改为统一格式
