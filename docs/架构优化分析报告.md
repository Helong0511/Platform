# services/product_service.py
class ProductService:
    @staticmethod
    async def get_product_by_id(
        product_id: int,
        user: User,
        db: AsyncSession
    ) -> Product:
        """获取产品详情（带权限检查）"""
        query = select(Product).options(selectinload(Product.client))
        
        # 权限过滤
        if user.role != UserRole.ADMIN:
            query = query.join(Client).where(Client.user_id == user.id)
        
        query = query.where(Product.id == product_id)
        result = await db.execute(query)
        product = result.scalars().first()
        
        if not product:
            if user.role != UserRole.ADMIN:
                raise_api_exception(ErrorCode.PRODUCT_NO_PERMISSION, ...)
            raise_api_exception(ErrorCode.PRODUCT_NOT_FOUND, ...)
        
        return product

# endpoints/products.py

    product_id: int,
    db: AsyncSession = Depends(get_db),

**优势：**
- ✅ 业务逻辑可复用

### 2.3 数据库查询优化

**问题：**
- 部分查询可能存在N+1问题
- 复杂查询缺少优化

**优化方案：**

#### 2.3.1 使用批量加载（Batch Loading）

```python
# 优化前
offers = await db.execute(select(Offer))
for offer in offers:
    # N+1查询：每个offer都查询一次advertiser
    advertiser = await db.execute(select(Advertiser).where(Advertiser.id == offer.advertiser_id))

# 优化后
from sqlalchemy.orm import selectinload, joinedload

query = select(Offer).options(
    joinedload(Offer.advertiser),  # 使用JOIN加载
    selectinload(Offer.offer_assignments)  # 批量加载关联数据
)
offers = await db.execute(query)
# 所有关联数据已加载，无额外查询
```

#### 2.3.2 使用查询优化器

```python
# 创建查询优化服务
class QueryOptimizer:
    @staticmethod
    def optimize_offer_query(
        query: Select,
        include_advertiser: bool = True,
        include_assignments: bool = False,
        include_caps: bool = False
    ) -> Select:
        """优化Offer查询，按需加载关联数据"""
        options = []
        
        if include_advertiser:
            options.append(joinedload(Offer.advertiser))
        if include_assignments:
            options.append(selectinload(Offer.offer_assignments))
        if include_caps:
            options.append(selectinload(Offer.caps))
        
        if options:
            query = query.options(*options)
        
        return query
```

#### 2.3.3 使用数据库视图（View）

对于复杂查询，可以创建数据库视图：

```sql
-- 创建转化统计视图
CREATE VIEW conversion_stats_view AS
SELECT 
    o.id as offer_id,
    o.offer_name,
    a.id as advertiser_id,
    a.advertiser_name,
    p.id as publisher_id,
    p.publisher_name,
    COUNT(pc.id) as conversion_count,
    SUM(pc.payout) as total_payout,
    DATE(pc.created_at) as conversion_date
FROM offers o
LEFT JOIN advertisers a ON o.advertiser_id = a.id
LEFT JOIN postback_conversions pc ON pc.offer_id = o.id
LEFT JOIN clicks c ON c.click_id = pc.click_id
LEFT JOIN publishers p ON p.id = c.publisher_id
GROUP BY o.id, a.id, p.id, DATE(pc.created_at);
```

### 2.4 完善缓存策略

**问题：**
- Redis使用场景有限（仅限流、短链接缓存）
- 缺少缓存失效策略
- 缺少缓存抽象层

**优化方案：**

#### 2.4.1 创建缓存抽象层

```python
# core/infrastructure/cache.py
from abc import ABC, abstractmethod
from typing import Optional, Any
import json

class CacheBackend(ABC):
    @abstractmethod
    async def get(self, key: str) -> Optional[Any]:
        pass
    
    @abstractmethod
    async def set(self, key: str, value: Any, ttl: Optional[int] = None):
        pass
    
    @abstractmethod
    async def delete(self, key: str):
        pass

class RedisCacheBackend(CacheBackend):
    def __init__(self, redis_client):
        self.redis = redis_client
    
    async def get(self, key: str) -> Optional[Any]:
        value = await self.redis.get(key)
        if value:
            return json.loads(value)
        return None
    
    async def set(self, key: str, value: Any, ttl: Optional[int] = None):
        await self.redis.set(key, json.dumps(value), ex=ttl)
    
    async def delete(self, key: str):
        await self.redis.delete(key)

# 缓存装饰器
def cached(key_pattern: str, ttl: int = 300):
    """缓存装饰器"""
    def decorator(func):
        async def wrapper(*args, **kwargs):
            # 生成缓存键
            cache_key = key_pattern.format(*args, **kwargs)
            
            # 尝试从缓存获取
            cache = get_cache_backend()
            cached_value = await cache.get(cache_key)
            if cached_value is not None:
                return cached_value
            
            # 执行函数
            result = await func(*args, **kwargs)
            
            # 写入缓存
            await cache.set(cache_key, result, ttl=ttl)
            return result
        return wrapper
    return decorator
```

class CacheInvalidator:
    """缓存失效管理器"""
    
    @staticmethod
    async def invalidate_offer_cache(offer_id: int):
        """失效Offer相关缓存"""
        cache = get_cache_backend()
        patterns = [
            f"offer:{offer_id}",
            f"offer:list:*",  # 列表缓存
            f"stats:*",  # 统计缓存
        ]
        for pattern in patterns:
            await cache.delete_pattern(pattern)
    
    @staticmethod
    async def invalidate_user_cache(user_id: int):
        """失效用户相关缓存"""
        cache = get_cache_backend()
        await cache.delete(f"user:{user_id}")
        await cache.delete_pattern(f"stats:dashboard:{user_id}:*")

### 2.5 引入仓储模式（Repository Pattern）

**问题：**
- 数据库查询逻辑分散在各处
- 难以统一管理查询逻辑

**解决方案：**

```python
# repositories/base_repository.py
from abc import ABC, abstractmethod
from typing import Generic, TypeVar, Optional, List
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.future import select

T = TypeVar('T')

class BaseRepository(ABC, Generic[T]):
    def __init__(self, db: AsyncSession, model: type[T]):
        self.db = db
        self.model = model
    
    async def get_by_id(self, id: int) -> Optional[T]:
        result = await self.db.execute(select(self.model).where(self.model.id == id))
        return result.scalar_one_or_none()
    
    async def get_all(self, skip: int = 0, limit: int = 100) -> List[T]:
        result = await self.db.execute(
            select(self.model).offset(skip).limit(limit)
        )
        return result.scalars().all()
    
    async def create(self, entity: T) -> T:
        self.db.add(entity)
        await self.db.commit()
        await self.db.refresh(entity)
        return entity
    
    async def update(self, entity: T) -> T:
        await self.db.commit()
        await self.db.refresh(entity)
        return entity
    
    async def delete(self, entity: T):
        await self.db.delete(entity)
        await self.db.commit()

# repositories/offer_repository.py
class OfferRepository(BaseRepository[Offer]):
    def __init__(self, db: AsyncSession):
        super().__init__(db, Offer)
    
    async def get_by_advertiser_id(self, advertiser_id: int) -> List[Offer]:
        result = await self.db.execute(
            select(Offer).where(Offer.advertiser_id == advertiser_id)
        )
        return result.scalars().all()
    
    async def get_active_offers(self) -> List[Offer]:
        result = await self.db.execute(
            select(Offer).where(
                Offer.is_active == True,
                Offer.status == 'Active'
            )
        )
        return result.scalars().all()
```

### 2.7 异步任务队列优化

**问题：**
- 点击记录队列实现简单
- 缺少任务优先级和重试机制

**优化方案：**

```python
# core/infrastructure/task_queue.py
from enum import IntEnum
from typing import Dict, Any, Callable
import asyncio

class TaskPriority(IntEnum):
    LOW = 3
    NORMAL = 2
    HIGH = 1
    CRITICAL = 0

class TaskQueue:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.queues = {
            TaskPriority.CRITICAL: "queue:critical",
            TaskPriority.HIGH: "queue:high",
            TaskPriority.NORMAL: "queue:normal",
            TaskPriority.LOW: "queue:low",
        }
    
    async def enqueue(
        self,
        task_name: str,
        task_data: Dict[str, Any],
        priority: TaskPriority = TaskPriority.NORMAL,
        max_retries: int = 3
    ):
        """加入任务队列"""
        task = {
            "name": task_name,
            "data": task_data,
            "retries": 0,
            "max_retries": max_retries,
            "created_at": datetime.now().isoformat()
        }
        queue_key = self.queues[priority]
        await self.redis.rpush(queue_key, json.dumps(task))
    
    async def dequeue(self, priority: TaskPriority) -> Optional[Dict[str, Any]]:
        """从队列取出任务（按优先级）"""
        # 按优先级顺序尝试取出
        for p in [TaskPriority.CRITICAL, TaskPriority.HIGH, TaskPriority.NORMAL, TaskPriority.LOW]:
            queue_key = self.queues[p]
            task_json = await self.redis.lpop(queue_key)
            if task_json:
                return json.loads(task_json)
        return None
```

# 在中间件中记录指标
class MetricsMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        start_time = time.time()
        response = await call_next(request)
        duration = time.time() - start_time

        request_count.labels(
            method=request.method,
            endpoint=request.url.path,
            status=response.status_code
